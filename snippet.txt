  };
  try{const raw=localStorage.getItem(DATA_KEY);if(raw)current=readState(raw)}catch(e){}
  let legacy=null;
  for(const key of LEGACY_KEYS){
    try{const raw=localStorage.getItem(key);if(!raw)continue;const x=readState(raw);if(x){legacy=x;break}}
    catch(e){}
  }
  // Main/legacy keys remain authoritative. Only if neither is usable do we search
  // other localStorage entries, and we choose the richest valid ledger rather than
  // the first matching object.
  if(!current&&!legacy){
    const candidates=[];
    try{
      for(let i=0;i<localStorage.length;i++){
        const key=localStorage.key(i);
        if(!key||key===DATA_KEY||LEGACY_KEYS.includes(key))continue;
        const raw=localStorage.getItem(key); if(!raw)continue;
        const x=readState(raw); if(x)candidates.push(x);
      }
    }catch(e){}
    candidates.sort((a,b)=>(b.transactions.length+b.programs.length)-(a.transactions.length+a.programs.length));
    current=candidates[0]||null;
  }
  if(!current)return legacy||defaultState();
  if(legacy){
    const programByName=new Map(current.programs.map(p=>[p.name.toLowerCase(),p]));
    legacy.programs.forEach(lp=>{
      if(!current.programs.some(p=>p.id===lp.id)){
        const sameName=programByName.get(lp.name.toLowerCase());
        if(!sameName){current.programs.push({...lp});programByName.set(lp.name.toLowerCase(),lp)}
      }
    });
    const ids=new Set(current.transactions.map(t=>t.id));
    legacy.transactions.forEach(lt=>{if(!ids.has(lt.id)){current.transactions.push({...lt});ids.add(lt.id)}});
    current.transactions.forEach(ct=>{
      const same=legacy.transactions.find(lt=>lt.date===ct.date&&Number(lt.miles)===Number(ct.miles)&&lt.type===ct.type&&lt.programId===ct.programId&&((lt.description||'')===(ct.description||'')));
      if(same&&ct.owner==='personal'&&same.owner==='business')ct.owner='business';
    });
  }
  if(!current.programs.length)current.programs=defaultState().programs;
  if(!current.programs.some(p=>!p.archived))current.programs[0].archived=false;
  const active=current.programs.find(p=>p.id===current.activeProgramId&&!p.archived)||current.programs.find(p=>!p.archived)||current.programs[0];
  if(active)current.activeProgramId=active.id;
  return current;
}
